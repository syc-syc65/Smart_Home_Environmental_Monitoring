1. OneNet_DevLink() - 连接OneNET服务器
----------------------------------------
**函数原型：**
```c
_Bool OneNet_DevLink(void);
```

**函数功能：**
- 与OneNET平台建立MQTT连接
- 处理TCP连接和MQTT协议握手
- 检查连接状态并返回结果

**实现代码：**
```c
_Bool OneNet_DevLink(void)
{
    MQTT_PACKET_STRUCTURE mqttPacket = {NULL, 0, 0, 0};  // 协议包
    unsigned char *dataPtr;
    _Bool status = 1;  // 默认失败
    
    // 1. 先关闭可能存在的旧连接
    ESP8266_CloseTCP();
    Delay_ms(200);
    
    // 2. 建立 TCP 连接到 OneNET 服务器
    if(!ESP8266_StartTCP(ONENET_SERVER_IP, ONENET_SERVER_PORT))
    {
        return 1;  // TCP 连接失败
    }
    Delay_ms(200);
    
    // 3. 生成 MQTT 连接数据包
    if(MQTT_PacketConnect(PROID, AUTH_INFO, DEVID, 256, 1, MQTT_QOS_LEVEL0, NULL, NULL, 0, &mqttPacket) == 0)
    {
        // 4. 发送 MQTT 连接数据包
        ESP8266_SendData(mqttPacket._data, mqttPacket._len);
        
        // 5. 等待平台响应（10秒超时）
        dataPtr = ESP8266_GetIPD(1000);
        if(dataPtr != NULL)
        {
            // 6. 解析 MQTT 响应
            if(MQTT_UnPacketRecv(dataPtr) == MQTT_PKT_CONNACK)
            {
                // 7. 检查连接确认码
                switch(MQTT_UnPacketConnectAck(dataPtr))
                {
                    case 0: status = 0; break;  // 连接成功
                    case 1: break;  // 协议版本不可接受
                    case 2: break;  // 标识符被拒绝
                    case 3: break;  // 服务器不可用
                    case 4: break;  // 用户名或密码错误
                    case 5: break;  // 未授权
                    default: break;
                }
            }
        }
        
        // 8. 释放 MQTT 数据包
        MQTT_DeleteBuffer(&mqttPacket);
    }
    
    return status;  // 0表示成功，1表示失败
}
```

**参数说明：**
- **PROID**: 产品ID (K3GGK1Sj7Q)
- **AUTH_INFO**: 鉴权Token
- **DEVID**: 设备名称 (stm)
- **ONENET_SERVER_IP**: OneNET服务器IP (mqtts.heclouds.com)
- **ONENET_SERVER_PORT**: OneNET MQTT端口 (1883)

**返回值：**
- 0: 连接成功
- 1: 连接失败

【小程序数据解析】

1. 解析传感器数据
----------------------------------------
```javascript
parseDeviceData(dataArray) {
  const sensorData = { ...this.data.sensorData }
  const controlData = { ...this.data.controlData }
  
  dataArray.forEach(item => {
    if (item.identifier === 'temperature' && item.value !== undefined) {
      sensorData.temperature = parseFloat(item.value).toFixed(1)
    } else if (item.identifier === 'humidity' && item.value !== undefined) {
      sensorData.humidity = parseFloat(item.value).toFixed(1)
    } else if (item.identifier === 'light' && item.value !== undefined) {
      sensorData.light = parseInt(item.value).toString()
    } else if (item.identifier === 'smoke' && item.value !== undefined) {
      sensorData.smoke = parseInt(item.value).toString()
    } else if (item.identifier === 'fan' && item.value !== undefined) {
      controlData.fan = this.parseBooleanValue(item.value)
    } else if (item.identifier === 'valve' && item.value !== undefined) {
      controlData.valve = this.parseBooleanValue(item.value)
    } else if (item.identifier === 'alarm' && item.value !== undefined) {
      controlData.alarm = this.parseBooleanValue(item.value)
    }
  })
  
  this.setData({ sensorData, controlData })
}
```

2. 解析布尔值
----------------------------------------
```javascript
parseBooleanValue(value) {
  if (typeof value === 'boolean') return value
  if (typeof value === 'string') {
    const lowerValue = value.toLowerCase().trim()
    return lowerValue === 'true' || lowerValue === '1' || lowerValue === 'yes'
  }
  if (typeof value === 'number') return value === 1
  return !!value
}
```

【小程序自动控制逻辑】

小程序支持两种控制模式：手动模式和自动模式

1. 手动模式
----------------------------------------

2. 自动模式
----------------------------------------
根据传感器数据自动控制设备：

```javascript
autoControl(sensorData, controlData) {
  const temperature = parseFloat(sensorData.temperature)
  const smoke = parseInt(sensorData.smoke)

  // 烟雾超标报警
  if (!isNaN(smoke) && smoke > 1500) {
    if (!controlData.alarm) {
      this.sendControlCommand('alarm', true, true)
    }
    if (!controlData.fan) {
      this.sendControlCommand('fan', true, true)
    }
  }

  // 温度过高降温
  if (!isNaN(temperature) && temperature > 35) {
    if (!controlData.fan) {
      this.sendControlCommand('fan', true, true)
    }
  }

  // 环境恢复正常
  if (!isNaN(temperature) && temperature < 33 && !isNaN(smoke) && smoke < 1400) {
    if (controlData.fan) {
      this.sendControlCommand('fan', false, true)
    }
    if (controlData.alarm) {
      this.sendControlCommand('alarm', false, true)
    }
  }
}
```
